% Literal author-numbered headings use the same numberline layout as generated
% headings. Match only a complete leading numeric indicator plus its separator;
% Exactly one ordinary separator space is accepted; repeated whitespace,
% unmatched titles and existing numberline entries pass through unchanged.

% The source chapter carries its literal numeric indicator inside chapter*.
% Give only such titles the reference's left alignment; unnumbered apparatus
% headings retain abntex2's centered appearance and original token lists.
\cs_new_eq:NN \memorial_saved_printchaptertitle:n \printchaptertitle
\cs_set_protected:Npn \printchaptertitle #1
  {
    \regex_match:nnTF { \A [0-9]+ \x{20} } {#1}
      { \chaptitlefont \raggedright \ABNTEXchapterupperifneeded{#1} }
      { \memorial_saved_printchaptertitle:n {#1} }
  }

\seq_new:N \l_memorial_toc_parts_seq
\tl_new:N \l_memorial_toc_number_tl
\tl_new:N \l_memorial_toc_title_tl
\cs_new_protected:Npn \memorial_toc_write:nnn #1#2#3
  {
    \str_if_eq:nnTF {#1} {toc}
      {
        \clist_if_in:nnTF {chapter,section,subsection} {#2}
          {
            \regex_extract_once:nnNTF
              { \A ([0-9]+ (?: \.[0-9]+ )* \.? (?: \x{20} - )?) \x{20} (\S.*) \Z }
              {#3} \l_memorial_toc_parts_seq
              {
                \seq_pop_left:NN \l_memorial_toc_parts_seq \l_memorial_toc_title_tl
                \seq_pop_left:NN \l_memorial_toc_parts_seq \l_memorial_toc_number_tl
                \seq_pop_left:NN \l_memorial_toc_parts_seq \l_memorial_toc_title_tl
                % Starred chapter hooks indent unnumbered entries in abntex2.
                % This entry has an explicit number, so restore its number column.
                \str_if_eq:nnT {#2} {chapter}
                  {
                    \addtocontents{toc}
                      {\protect\cftsetindents{chapter}{0pt}{\protect\cftlastnumwidth}}
                  }
                \use:x
                  {
                    \exp_not:N \memorial_saved_addcontentsline:nnn {#1}{#2}
                      {
                        \exp_not:N \protect \exp_not:N \numberline
                          {\exp_not:V \l_memorial_toc_number_tl}
                        \exp_not:V \l_memorial_toc_title_tl
                      }
                  }
              }
              { \memorial_saved_addcontentsline:nnn {#1}{#2}{#3} }
          }
          { \memorial_saved_addcontentsline:nnn {#1}{#2}{#3} }
      }
      { \memorial_saved_addcontentsline:nnn {#1}{#2}{#3} }
  }
