% Reflection boxes contain ordinary paragraphs, not floats or nested boxes.
% The collected vertical list is divided at TeX paragraph/line breakpoints.
% Each segment receives a border; original wording is never changed to fit.
\newbox\ReflectionBody
\newbox\ReflectionSegment
\newdimen\ReflectionAvailable
\newdimen\ReflectionWidth
\newdimen\ReflectionTotal
% Source leading tabs are retained as reconstructible tokens. The approved
% reference layout uses one paragraph indent independently of their count.
\newcommand{\originaltab}{\hspace*{0pt}}
% Leading source spaces likewise remain explicit without adding a second
% indentation beside the paragraph indent. Interior/trailing spaces keep
% originalspace below and retain their ordinary visible width.
\newcommand{\originalleadingspace}{\hspace*{0pt}}
% Preserve a source ODF prefix line-break token without adding another line;
% the preceding figure or paragraph already supplies the structural break.
\newcommand{\originalleadbreak}{\hspace*{0pt}}
% Authorized formatting correction: an internal source tab remains explicit
% in the unit while rendering as one normal interword space.
\newcommand{\originalinlinetab}{\unskip\space}
% Explicit source space survives at paragraph/title edges.
\newcommand{\originalspace}{\hspace*{\fontdimen2\font}}
\newcommand{\ReflectionRender}{%
  \ifvoid\ReflectionBody\else
    \ReflectionAvailable=\pagegoal
    \ifdim\ReflectionAvailable>\textheight
      \ReflectionAvailable=\textheight
    \fi
    \advance\ReflectionAvailable by -\pagetotal
    \advance\ReflectionAvailable by -2\fboxsep
    \advance\ReflectionAvailable by -2\fboxrule
    \advance\ReflectionAvailable by -\baselineskip
    \ifdim\ReflectionAvailable<4\baselineskip
      \newpage
      \ReflectionAvailable=\textheight
      \advance\ReflectionAvailable by -2\fboxsep
      \advance\ReflectionAvailable by -2\fboxrule
      \advance\ReflectionAvailable by -\baselineskip
    \fi
    \ReflectionTotal=\ht\ReflectionBody
    \advance\ReflectionTotal by \dp\ReflectionBody
    \ifdim\ReflectionTotal>\ReflectionAvailable
      \splittopskip=\topskip
      \splitmaxdepth=\maxdepth
      % vsplit's fixed-height glue is immediately discarded by natural-height
      % reboxing; badness of that temporary target is not a layout defect.
      % All horizontal/output diagnostics retain their normal thresholds.
      \vbadness=10000
      \setbox\ReflectionSegment=\vsplit\ReflectionBody to \ReflectionAvailable
      \vbadness=1000
      % Natural-height reboxing removes the target-height stretch.
      \setbox\ReflectionSegment=\vbox{\unvbox\ReflectionSegment}
      \noindent\fbox{\box\ReflectionSegment}\par
      \newpage
      \expandafter\ReflectionRender
    \else
      \noindent\fbox{\box\ReflectionBody}\par
    \fi
  \fi
}
\newenvironment{reflection}{%
  % Force the page builder to consume preceding paragraph lines before
  % measuring pagegoal/pagetotal; penalty zero permits, never forces, a break.
  \par\penalty0\begingroup
  \setlength{\fboxsep}{6pt}\setlength{\fboxrule}{0.4pt}
  \ReflectionWidth=\linewidth
  \advance\ReflectionWidth by -2\fboxsep
  \advance\ReflectionWidth by -2\fboxrule
  \setbox\ReflectionBody=\vbox\bgroup
    \hsize=\ReflectionWidth\linewidth=\ReflectionWidth
    \normalfont\normalsize\sffamily
}{%
    \par\egroup
    \ReflectionRender
  \endgroup\par
}

% Captions already carry the original literal "Figure N:" identifier.
% caption* prevents an extra generated prefix. ODF paragraph styles P64/P95
% inherit italics through Figure -> Caption; graphic styles T50/T51/T71/T141
% add no italic override. The requested output explicitly uses upright shape
% for the caption and LoF entry while preserving the caption wording.
% The counter and label still provide live cross-references.
% The locked engine is XeTeX. Its native reader with an explicit display width
% handles original JPEG EXIF zero dimensions without rewriting image bytes.
% AGES I assets are landscape images; full text width fits the page height.
% Arguments: intact asset, literal caption, unique label, source apparatus,
% width as a fraction of linewidth. Unknown sources must be declared pending.
\newcommand{\memorialfigure}[5]{%
  \begin{figure}[htbp]
    \refstepcounter{figure}%
    \caption*{{\normalfont\upshape #2}}\label{#3}%
    \addcontentsline{lof}{figure}{{\normalfont\upshape #2}}%
    \noindent\makebox[\linewidth][c]{\XeTeXpicfile "#1" width #5\linewidth\relax}%
    \par\small\SingleSpacing #4\par
  \end{figure}
}
